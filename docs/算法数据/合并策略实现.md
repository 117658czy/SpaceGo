# 合并策略实现逻辑

## 1. 核心设计概述

合并策略是HMGS-DP引擎的核心组件之一，负责判断节点间是否应该合并，以及如何合并。它基于分层精度的CNN合并函数和合并决策的UCT框架，实现了高效、准确的节点合并，减少重复计算，提高搜索效率。

## 3. 核心算法实现

### 3.1 分层精度的CNN合并函数

```python
def hierarchical_cnn_merge(self, node1: HMGSNode, node2: HMGSNode) -> float:
    """分层精度的CNN合并函数"""
    # 确定节点深度
    max_depth = max(node1.depth, node2.depth)
    # 根据深度选择CNN精度等级
    cnn_accuracy_level = self._select_cnn_accuracy_level(max_depth)
    # 使用选定的精度等级进行合并判断
    merge_similarity = self._cnn_merge_with_accuracy_level(node1, node2, cnn_accuracy_level)
    return merge_similarity
```

**分层策略：**
- **浅层节点（深度 < 3）**：使用轻量CNN，低分辨率表示，快速计算
- **中层节点（3 ≤ 深度 < 10）**：使用标准CNN，中等分辨率，平衡精度与速度
- **深层节点（深度 ≥ 10）**：使用高精度CNN，完整分辨率，确保准确性

### 3.2 合并质量评分

```python
def calculate_merge_score(self, node1: HMGSNode, node2: HMGSNode) -> float:
    """计算合并质量评分"""
    # 计算状态相似性
    state_similarity = self._attention_enhanced_merge_judgment(node1, node2)
    # 计算价值兼容性
    value_compatibility = self.calculate_value_compatibility(node1, node2)
    # 计算拓扑一致性
    topology_consistency = self.calculate_topology_consistency(node1, node2)
    # 加权求和
    merge_score = (self.cosine_similarity_weight * state_similarity +
                  self.value_compatibility_weight * value_compatibility +
                  self.topology_consistency_weight * topology_consistency)
    return merge_score
```

### 3.3 状态相似性计算

#### 3.3.1 注意力增强的合并判断
```python
def _attention_enhanced_merge_judgment(self, node1: HMGSNode, node2: HMGSNode) -> float:
    """注意力增强的合并判断"""
    # 获取节点状态编码
    state1 = node1.state_encoding or self._generate_default_encoding(node1)
    state2 = node2.state_encoding or self._generate_default_encoding(node2)
    # 计算注意力权重
    attention_weight1 = self._calculate_attention_weight(state1, state2)
    attention_weight2 = self._calculate_attention_weight(state2, state1)
    # 应用注意力权重
    attended_state1 = [v * attention_weight1 for v in state1]
    attended_state2 = [v * attention_weight2 for v in state2]
    # 计算相似度
    similarity = self._cosine_similarity(attended_state1, attended_state2)
    return (similarity + 1) / 2  # 映射到0-1范围
```

#### 3.3.2 基础状态相似性
```python
def _basic_state_similarity(self, node1: HMGSNode, node2: HMGSNode) -> float:
    """基础状态相似性计算"""
    # 获取棋盘状态
    board1 = node1.state.get_board()
    board2 = node2.state.get_board()
    # 计算相同位置比例
    total_cells = board1.size
    same_cells = (board1 == board2).sum()
    return same_cells / total_cells  # 归一化到0-1范围
```

### 3.4 价值兼容性计算

```python
def calculate_value_compatibility(self, node1: HMGSNode, node2: HMGSNode) -> float:
    """计算价值兼容性"""
    # 计算平均价值
    value1 = node1.total_value / node1.visit_count if node1.visit_count > 0 else 0.0
    value2 = node2.total_value / node2.visit_count if node2.visit_count > 0 else 0.0
    # 计算价值差的绝对值，归一化
    value_diff = abs(value1 - value2)
    max_value_diff = 2.0  # 假设价值范围是[-1, 1]
    compatibility = 1.0 - min(value_diff / max_value_diff, 1.0)
    return compatibility
```

### 3.5 拓扑一致性计算

```python
def calculate_topology_consistency(self, node1: HMGSNode, node2: HMGSNode) -> float:
    """计算拓扑一致性"""
    # 使用缓存避免重复计算
    node_ids = tuple(sorted([id(node1), id(node2)]))
    if node_ids in self._topology_consistency_cache:
        return self._topology_consistency_cache[node_ids]
    
    # 计算合并概率和被合并概率的相似度
    merge_prob_diff = abs(node1.merge_prob - node2.merge_prob)
    merged_into_prob_diff = abs(node1.merged_into_prob - node2.merged_into_prob)
    
    # 归一化到0-1范围
    merge_prob_consistency = 1.0 - merge_prob_diff
    merged_into_prob_consistency = 1.0 - merged_into_prob_diff
    
    # 计算平均一致性
    consistency = (merge_prob_consistency + merged_into_prob_consistency) / 2
    
    # 保存到缓存
    self._topology_consistency_cache[node_ids] = consistency
    # 限制缓存大小
    if len(self._topology_consistency_cache) > 1000:
        self._topology_consistency_cache.pop(next(iter(self._topology_consistency_cache)))
    
    return consistency
```

### 3.6 合并UCT框架

```python
def calculate_merge_uct(self, node: HMGSNode, candidate: HMGSNode, parent: HMGSNode, c_merge: float = 1.0) -> float:
    """计算合并决策的UCT值"""
    # 计算合并质量评分
    merge_score = self.calculate_merge_score(node, candidate)
    # 计算合并的历史成功率
    success_rate = 1.0
    if hasattr(self, 'success_count') and hasattr(self, 'total_count') and self.total_count > 0:
        success_rate = self.success_count / self.total_count
    # 计算探索项
    parent_visits = math.sqrt(parent.visit_count)
    candidate_merges = candidate.visit_count
    # 合并UCT公式：M-UCT(merge_action) = Q_merge + C_merge × √(ln N_merge_parent / N_merge_action)
    q_merge = merge_score * success_rate
    exploration_term = c_merge * math.sqrt(math.log(max(1, parent.visit_count)) / (1 + candidate_merges))
    # 考虑合并的计算成本
    merge_cost = (len(node.children) + len(candidate.children)) / 100.0
    # 最终合并UCT值
    merge_uct = q_merge + exploration_term - merge_cost
    return merge_uct
```

### 3.7 节点合并实现

```python
def merge_nodes(self, node1: HMGSNode, node2: HMGSNode) -> HMGSNode:
    """合并两个节点"""
    # 创建新的合并节点
    merged_state = node1.state.clone()  # 使用node1的状态作为基础
    merged_node = HMGSNode(merged_state, node1.parent)
    
    # 合并访问次数和总价值（加权平均）
    total_visits = node1.visit_count + node2.visit_count
    merged_node.visit_count = total_visits
    
    # 避免除以零错误
    if total_visits == 0:
        merged_node.total_value = (node1.total_value + node2.total_value) / 2
        merged_node.merge_prob = (node1.merge_prob + node2.merge_prob) / 2
        merged_node.merged_into_prob = (node1.merged_into_prob + node2.merged_into_prob) / 2
    else:
        merged_node.total_value = (node1.total_value + node2.total_value) / 2
        merged_node.merge_prob = (node1.merge_prob * node1.visit_count + node2.merge_prob * node2.visit_count) / total_visits
        merged_node.merged_into_prob = (node1.merged_into_prob * node1.visit_count + node2.merged_into_prob * node2.visit_count) / total_visits
    
    # 合并子节点
    merged_node.children = node1.children + node2.children
    merged_node.moves = list(set(node1.moves + node2.moves))
    
    # 合并三副本数据
    for i in range(3):
        merged_node.replica_visits[i] = node1.replica_visits[i] + node2.replica_visits[i]
        merged_node.replica_values[i] = node1.replica_values[i] + node2.replica_values[i]
    
    # 更新合并统计信息
    node1.update_merge_stats(True)
    node2.update_merge_stats(True, merged_into=True)
    merged_node.merged_nodes = [node1, node2]
    
    return merged_node
```

```python
def calculate_batch_merge_score(self, nodes1: List[HMGSNode], nodes2: List[HMGSNode]) -> np.ndarray:
    """批量计算合并质量评分"""
    batch_size = len(nodes1)
    merge_scores = np.zeros(batch_size, dtype=np.float32)
    for i in range(batch_size):
        merge_scores[i] = self.calculate_merge_score(nodes1[i], nodes2[i])
    return merge_scores
```

### 5.2 并行处理
```python
def parallel_get_merge_candidates(self, nodes: List[HMGSNode], all_nodes: List[HMGSNode], max_workers: int = None) -> List[List[Tuple[HMGSNode, float]]]:
    """并行获取多个节点的合并候选"""
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = [executor.submit(self.get_merge_candidates, node, all_nodes) for node in nodes]
        results = [future.result() for future in futures]
    return results
```

## 7. 关键算法流程

### 7.1 合并决策流程

```
节点1, 节点2
├─ 分层精度的CNN合并函数
│  ├─ 确定节点深度
│  ├─ 选择CNN精度等级
│  │  ├─ 浅层节点 → 轻量CNN
│  │  ├─ 中层节点 → 标准CNN
│  │  └─ 深层节点 → 高精度CNN
│  └─ 计算状态相似性
├─ 计算价值兼容性
├─ 计算拓扑一致性
├─ 计算合并质量评分
├─ 计算合并UCT值
│  ├─ 计算合并质量评分
│  ├─ 计算探索项
│  └─ 考虑合并成本
└─ 决定是否合并
   ├─ 合并节点
   └─ 调整权重
```

### 7.2 分层CNN合并流程

```
输入节点1, 节点2
├─ 计算节点深度
├─ 选择CNN精度等级
│  ├─ 轻量CNN → 低分辨率表示
│  ├─ 标准CNN → 中等分辨率表示
│  └─ 高精度CNN → 完整分辨率表示
├─ 调整状态表示分辨率
│  ├─ 轻量CNN：降低50%分辨率
│  ├─ 标准CNN：降低20%分辨率
│  └─ 高精度CNN：保持完整分辨率
├─ 计算状态编码相似性
│  ├─ 轻量CNN：欧氏距离
│  └─ 高精度CNN：余弦相似度
└─ 返回合并相似度
```
