# 神经网络集成实现

## 核心设计

MiniGo AI系统采用了多种先进的神经网络架构，通过集成不同类型的神经网络模型，实现了高效的策略预测和价值评估。神经网络集成是系统的核心组件之一，负责将棋盘状态转换为策略分布和价值估计，为搜索算法提供指导。

- **多种网络架构**：支持EnhancedNeuralNetwork、StyleAdaptedNN和HybridNeuralNetwork三种主要架构
- **模块化设计**：采用模块化组件，便于扩展和组合
- **注意力机制**：多种注意力头设计，关注不同维度的棋局信息
- **多尺度特征提取**：同时关注局部、中距离和全局模式
- **Transformer增强**：结合Transformer架构，增强对全局关系的建模能力
- **图神经网络集成**：利用GCN处理棋盘的图结构关系
- **专业化头设计**：多个专业化头，分别负责不同的预测任务

## 核心组件

### 残差块

```python
class ResBlock(nn.Module):
    """残差块，用于神经网络的主要结构"""
    def __init__(self, in_channels, out_channels):
        super(ResBlock, self).__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1)
        self.bn1 = nn.BatchNorm2d(out_channels)
        self.conv2 = nn.Conv2d(out_channels, out_channels, kernel_size=3, padding=1)
        self.bn2 = nn.BatchNorm2d(out_channels)
        self.relu = nn.ReLU()
        
        # 快捷连接，如果输入输出通道不同则使用1x1卷积调整
        self.shortcut = nn.Sequential()
        if in_channels != out_channels:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1),
                nn.BatchNorm2d(out_channels)
            )
    
    def forward(self, x):
        residual = x
        x = self.relu(self.bn1(self.conv1(x)))
        x = self.bn2(self.conv2(x))
        x += self.shortcut(residual)
        x = self.relu(x)
        return x
```

```python
class ConnectionAttention(nn.Module):
    def __init__(self, num_channels, nhead=4):
        super().__init__()
```

```python
class GCNLayer(nn.Module):
    """图卷积网络层"""
    def __init__(self, in_features, out_features):
        super(GCNLayer, self).__init__()
        self.linear = nn.Linear(in_features, out_features)
        self.relu = nn.ReLU()
    
    def forward(self, x, adj_matrix):
        # x: [batch_size, num_nodes, in_features]
        # adj_matrix: [batch_size, num_nodes, num_nodes]
        support = self.linear(x)
        output = torch.bmm(adj_matrix, support)
        return self.relu(output)
```

```python
class SpatialAttentionHead(nn.Module):
    """空间策略头：预测棋盘位置概率分布"""
    def __init__(self, in_channels, board_size):
        super(SpatialAttentionHead, self).__init__()
        self.conv = nn.Conv2d(in_channels, 2, kernel_size=1)
        self.bn = nn.BatchNorm2d(2)
        self.relu = nn.ReLU()
        self.fc = nn.Linear(2 * board_size * board_size, board_size * board_size)
        self.board_size = board_size
    
    def forward(self, x):
        x = self.relu(self.bn(self.conv(x)))
        x = x.view(-1, 2 * self.board_size * self.board_size)
        x = self.fc(x)
        x = F.softmax(x, dim=1)
        return x

class GlobalPolicyHead(nn.Module):
    """全局策略头：预测Pass和其他全局动作"""
    def __init__(self, in_channels, board_size):
        super(GlobalPolicyHead, self).__init__()
        self.conv = nn.Conv2d(in_channels, 1, kernel_size=1)
        self.bn = nn.BatchNorm2d(1)
        self.relu = nn.ReLU()
        self.fc = nn.Linear(board_size * board_size, 5)  # 5种全局动作（包括pass）
    
    def forward(self, x):
        x = self.relu(self.bn(self.conv(x)))
        x = x.view(-1, x.size(1) * x.size(2) * x.size(3))
        x = self.fc(x)
        x = F.softmax(x, dim=1)
        return x

class LocalPolicyHead(nn.Module):
    """局部策略头：预测特定区域的局部战斗"""
    def __init__(self, in_channels, board_size):
        super(LocalPolicyHead, self).__init__()
        self.conv = nn.Conv2d(in_channels, 2, kernel_size=3, padding=1)
        self.bn = nn.BatchNorm2d(2)
        self.relu = nn.ReLU()
        self.fc = nn.Linear(2 * board_size * board_size, board_size * board_size)
        self.board_size = board_size
    
    def forward(self, x):
        x = self.relu(self.bn(self.conv(x)))
        x = x.view(-1, 2 * self.board_size * self.board_size)
        x = self.fc(x)
```

#### 价值头

```python
class EnhancedValueHead(nn.Module):
    """增强版价值头：结合胜负预测和形势判断"""
    def __init__(self, in_channels, board_size):
        super(EnhancedValueHead, self).__init__()
        self.board_size = board_size
        
        # 多尺度特征提取
        self.conv1 = nn.Conv2d(in_channels, 2, kernel_size=1)
        self.conv3 = nn.Conv2d(in_channels, 2, kernel_size=3, padding=1)
        self.conv5 = nn.Conv2d(in_channels, 2, kernel_size=5, padding=2)
        self.bn = nn.BatchNorm2d(6)
        self.relu = nn.ReLU()
        self.dropout = nn.Dropout(0.4)
        
        # 全连接层
        self.fc1 = nn.Linear(6 * board_size * board_size, 512)
        self.fc2 = nn.Linear(512, 256)
        self.fc3 = nn.Linear(256, 128)
        self.fc4 = nn.Linear(128, 1)
        
        # 残差连接
        self.res_fc = nn.Linear(6 * board_size * board_size, 1)
    
    def forward(self, x):
        # 多尺度特征提取
        x1 = self.conv1(x)
        x3 = self.conv3(x)
        x5 = self.conv5(x)
        x = torch.cat([x1, x3, x5], dim=1)
        x = self.relu(self.bn(x))
        x = self.dropout(x)
        
        # 展平
        x_flat = x.view(-1, 6 * self.board_size * self.board_size)
        
        # 残差连接
        res = self.res_fc(x_flat)
        
        # 深层全连接网络
        x = self.relu(self.fc1(x_flat))
        x = self.dropout(x)
        x = self.relu(self.fc2(x))
        x = self.dropout(x)
        x = self.relu(self.fc3(x))
        x = self.fc4(x)
        
        # 结合残差连接
        x += res
        
        # 使用tanh激活函数将输出限制在[-1, 1]之间
        x = torch.tanh(x)
        return x
```

### 多网络架构集成

系统支持多种神经网络架构，可以根据不同场景和需求选择合适的网络：

- **EnhancedNeuralNetwork**：基础增强版网络，适合大多数场景

### 多尺度特征融合

```python
class MultiScaleAttention(nn.Module):
    """
    多尺度注意力模块
    能够同时关注局部、中距离和全局模式
    """
    def __init__(self, num_channels, board_size):
        super().__init__()
        self.num_channels = num_channels
        self.board_size = board_size
        
        # 局部注意力头（关注3x3区域）
        self.local_attention = nn.MultiheadAttention(num_channels, 4)
        # 中距离注意力头（关注7x7区域）
        self.medium_attention = nn.MultiheadAttention(num_channels, 4)
        # 全局注意力头（关注整个棋盘）
        self.global_attention = nn.MultiheadAttention(num_channels, 4)
        
        # 池化层用于不同尺度
        self.local_pool = nn.AvgPool2d(kernel_size=3, stride=1, padding=1)
        self.medium_pool = nn.AvgPool2d(kernel_size=7, stride=1, padding=3)
        
        # 融合层
        self.fusion_conv = nn.Conv2d(num_channels * 3, num_channels, kernel_size=1)
        self.fusion_bn = nn.BatchNorm2d(num_channels)
        self.relu = nn.ReLU()
    
    def forward(self, x):
        batch_size, channels, height, width = x.size()
        seq_len = height * width
        
        # 准备不同尺度的输入
        features = x.view(batch_size, channels, seq_len).permute(2, 0, 1)
        
        # 局部尺度特征（3x3）
        local_features = self.local_pool(x)
        local_features = local_features.view(batch_size, channels, seq_len).permute(2, 0, 1)
        
        # 中距离尺度特征（7x7）
        medium_features = self.medium_pool(x)
        medium_features = medium_features.view(batch_size, channels, seq_len).permute(2, 0, 1)
        
        # 应用不同尺度的注意力
        local_attn_output, _ = self.local_attention(features, local_features, local_features)
        local_attn_output = local_attn_output.permute(1, 2, 0).view(batch_size, channels, height, width)
        
        medium_attn_output, _ = self.medium_attention(features, medium_features, medium_features)
        medium_attn_output = medium_attn_output.permute(1, 2, 0).view(batch_size, channels, height, width)
        
        global_attn_output, _ = self.global_attention(features, features, features)
        global_attn_output = global_attn_output.permute(1, 2, 0).view(batch_size, channels, height, width)
        
        # 融合不同尺度的注意力输出
        fused = torch.cat([local_attn_output, medium_attn_output, global_attn_output], dim=1)
        fused = self.relu(self.fusion_bn(self.fusion_conv(fused)))
        
        # 残差连接
        return x + fused
```

## 输入输出处理

### 棋盘状态到张量转换

```python
def _board_to_tensor(self, board_state, player):
    """将棋盘状态转换为17通道的输入张量"""
    # 假设board_state是numpy数组，0:空，1:白，2:黑
    size = board_state.shape[0]
    
    # 创建17通道的输入张量
    input_tensor = np.zeros((1, 17, size, size), dtype=np.float32)
    
    # 通道1-8: 当前玩家的棋子（最近8步）
    # 这里简化处理，只使用当前状态的棋子
    input_tensor[0, 0, :, :] = (board_state == player).astype(np.float32)
    for i in range(1, 8):
        input_tensor[0, i, :, :] = input_tensor[0, 0, :, :] * 0.9  # 稍微降低权重模拟历史衰减
    
    # 通道9-16: 对手的棋子（最近8步）
    opponent = 1 if player == 2 else 2
    input_tensor[0, 8, :, :] = (board_state == opponent).astype(np.float32)
    for i in range(9, 16):
        input_tensor[0, i, :, :] = input_tensor[0, 8, :, :] * 0.9  # 稍微降低权重模拟历史衰减
    
    # 通道17: 玩家指示符（当前玩家为1，否则为0）
    input_tensor[0, 16, :, :] = 1.0 if player == 2 else 0.0  # 假设黑棋为1，白棋为0
    
    # 转换为PyTorch张量
    return torch.from_numpy(input_tensor)
```

### 预测结果处理

```python
def predict(self, board_state, player):
    """
    预测给定棋盘状态的策略和价值
    """
    with torch.no_grad():
        tensor = self._board_to_tensor(board_state, player)
        outputs = self.forward(tensor)
        
        # 组合多个策略头的输出
        combined_policy = outputs['spatial_policy'].cpu().numpy().flatten()
        
        # 提取价值预测
        value = outputs['win_prediction'].item()
        score = outputs['score_estimation'].item()
        safety = outputs['safety_evaluation'].cpu().numpy().reshape(self.board_size, self.board_size)
        
        # 提取链接神经元输出
        prior_prob = outputs['prior_prob'].cpu().numpy().flatten()
        value_estimate = outputs['value_estimate'].item()
        state_encoding = outputs['state_encoding'].cpu().numpy().flatten()
        uncertainty = outputs['uncertainty'].item()
        
        # 返回包含所有信息的字典
        result = {
            'policy': combined_policy,
            'value': value,
            'score': score,
            'safety': safety,
            'prior_prob': prior_prob,
            'value_estimate': value_estimate,
            'state_encoding': state_encoding,
            'uncertainty': uncertainty
        }
        
        return result
```

```python
                     attention_pred_weight=0.1, attention_consistency_weight=0.1, 
                     attention_mask_weight=0.05):
    """获取多目标学习的损失函数，包含注意力正则化和注意力相关辅助任务"""
    # 主要目标：策略和价值预测
    policy_loss = nn.CrossEntropyLoss()
    value_loss = nn.MSELoss()
    
    # 辅助目标
    opponent_move_loss = nn.CrossEntropyLoss()  # 对方落子预测
    life_death_loss = nn.BCELoss()              # 死活判断
    territory_loss = nn.MSELoss()               # 领地预测
    ladder_loss = nn.BCELoss()                  # 征子判断
    attention_pred_loss = nn.MSELoss()          # 注意力预测损失
    attention_consistency_loss = nn.MSELoss()    # 注意力一致性损失
    attention_mask_loss = nn.BCELoss()          # 注意力掩码损失
    
    # 注意力正则化函数
    def attention_regularization(attention_weights):
        """
        注意力正则化，防止注意力过度集中
        使用熵正则化，鼓励更均匀的注意力分布
        """
        if attention_weights is None:
            return 0.0
        # 计算注意力权重的熵
        entropy = -torch.sum(attention_weights * torch.log(attention_weights + 1e-10), dim=1)
        # 熵最大化，即最小化负熵
        return -entropy.mean()
    
    # 完整的损失函数
    def total_loss(outputs, targets):
        # 主要损失
        main_policy_loss = policy_loss(outputs['spatial_policy'], targets['policy'])
        main_value_loss = value_loss(outputs['win_prediction'], targets['value'])
        
        # 辅助损失
        opponent_loss = opponent_move_loss(outputs['opponent_move'], targets['opponent_move'])
        life_death_loss_val = life_death_loss(outputs['life_death'], targets['life_death'])
        territory_loss_val = territory_loss(outputs['territory'], targets['territory'])
        ladder_loss_val = ladder_loss(outputs['ladder'], targets['ladder'])
        
        # 注意力相关损失
        attention_reg = attention_regularization(outputs['attention_weights']) * attention_reg_weight
        attention_pred = attention_pred_loss(outputs['attention_pred'], targets['attention_target']) * attention_pred_weight
        attention_consistency = attention_consistency_loss(outputs['spatial_attention'], outputs['connection_attention']) * attention_consistency_weight
        attention_mask = attention_mask_loss(outputs['attention_mask'], targets['attention_mask']) * attention_mask_weight
        
        # 总损失 = 主要损失 + 辅助损失 + 注意力相关损失
        total = (main_policy_loss + main_value_loss) * main_weight +
                opponent_loss * opponent_move_weight +
                life_death_loss_val * life_death_weight +
                territory_loss_val * territory_weight +
                ladder_loss_val * ladder_weight +
                attention_reg +
                attention_pred +
                attention_consistency +
                attention_mask
        
        return total
    
    return total_loss
```

### 早退机制

在训练过程中，对于简单局面，网络会提前返回，提高训练效率：

```python
# 早期局面评估，用于早退机制
if self.training and torch.rand(1).item() < 0.1:  # 10%的概率进行早退检查
    early_value = torch.sigmoid(self.score_estimation_head(x))
    if torch.abs(early_value).mean() > self.easy_position_threshold:  # 简单局面阈值
        # 简单局面，提前返回
        dummy_policy = torch.zeros(batch_size, height * width, device=self.device)
        dummy_global_policy = torch.zeros(batch_size, 5, device=self.device)
        dummy_local_policy = torch.zeros(batch_size, height * width, device=self.device)
        dummy_state_encoding = torch.zeros(batch_size, self.num_channels, device=self.device)
        return {
            'spatial_policy': dummy_policy,
            'global_policy': dummy_global_policy,
            'local_policy': dummy_local_policy,
            'win_prediction': early_value,
            'score_estimation': early_value * 100,
            'safety_evaluation': torch.zeros(batch_size, height * width, device=self.device),
            'prior_prob': dummy_policy,
            'value_estimate': early_value,
            'state_encoding': dummy_state_encoding,
            'uncertainty': torch.zeros(batch_size, 1, device=self.device)
        }
```
