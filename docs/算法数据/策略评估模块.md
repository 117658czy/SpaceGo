# 策略评估模块实现

## 核心设计

策略评估模块是MiniGo AI系统的重要组成部分，负责评估不同AI策略的性能，提供全面的性能指标和分析报告。它支持多种评估指标和策略比较方法，帮助开发者理解和改进AI策略。

- **多维度评估指标**：支持胜率、平均得分、平均时间、探索率、利用效率等多种指标
- **基于胜率的积分系统**：实现类似Elo的积分计算，考虑对手强度和游戏局数
- **阶段评估**：分别评估开局、中盘和官子阶段的性能
- **策略比较**：支持多策略间的对比分析
- **详细性能报告**：生成包含多种统计数据和图表的性能报告
- **游戏模式分析**：自动识别和分类游戏模式
- **策略优化**：支持基于评估结果的策略参数优化

### WinRatePointsEvaluator类

```python
class WinRatePointsEvaluator:
    """
    基于胜率的积分评价器，用于计算AI对弈的积分评价
    """
    
    def __init__(self, base_points: int = 1000):
        """
        初始化基于胜率的积分评价器
        
        Args:
            base_points: 基础积分
        """
        self.base_points = base_points
        # 积分系数
        self.points_coefficients = {
            'win_rate_coeff': 2000,          # 胜率调整系数
            'opponent_strength_coeff': 0.01,  # 对手强度调整系数
            'game_count_coeff': 10,          # 局数调整系数
        }
        # 阶段定义
        self.game_phases = {
            'opening': {'start': 0, 'end': 30},    # 开局：0-30手
            'middle': {'start': 31, 'end': 150},   # 中盘：31-150手
            'endgame': {'start': 151, 'end': float('inf')} # 官子：151手以上
        }
```

## 核心评估功能

### 基于胜率的积分计算

实现类似Elo的积分系统，考虑对手强度和游戏局数：

```python
def calculate_points(self, win_rate: float, opponent_points: int = None, game_count: int = 1) -> float:
    """
    计算基于胜率的积分
    
    Args:
        win_rate: 胜率 (0-1)
        opponent_points: 对手积分
        game_count: 对弈局数
        
    Returns:
        计算得到的积分
    """
    opponent_points = opponent_points or self.base_points
    
    # 胜率调整分：将胜率转换为积分差
    win_rate_adjustment = (win_rate - 0.5) * self.points_coefficients['win_rate_coeff']
    
    # 对手强度调整分：考虑对手的积分水平
    opponent_strength_adjustment = (opponent_points - self.base_points) * \
                                   self.points_coefficients['opponent_strength_coeff'] * win_rate
    
    # 局数调整分：局数越多，积分越稳定
    import math
    game_count_adjustment = math.log10(game_count + 1) * self.points_coefficients['game_count_coeff']
    
    # 计算总积分，确保积分不为负数
    total_points = self.base_points + win_rate_adjustment + opponent_strength_adjustment + game_count_adjustment
    
    return max(0, total_points)
```

### 阶段积分计算

分别计算开局、中盘和官子阶段的积分：

```python
def calculate_phase_points(self, game_details: List[Dict], phase: str, opponent_points: int = None) -> float:
    """
    计算特定阶段的胜率积分
    
    Args:
        game_details: 游戏详情列表
        phase: 阶段名称 (opening, middle, endgame)
        opponent_points: 对手积分
        
    Returns:
        阶段积分
    """
    if phase not in self.game_phases:
        raise ValueError(f"无效的阶段名称: {phase}")
    
    phase_config = self.game_phases[phase]
    
    # 计算该阶段的胜率
    phase_wins = 0
    phase_total = 0
    
    for game in game_details:
        moves = game.get('moves', 0)
        if phase_config['start'] <= moves <= phase_config['end']:
            phase_total += 1
            # 检查是否获胜
            if (game['winner'] == 'black' and game['is_strategy_black']) or \
               (game['winner'] == 'white' and not game['is_strategy_black']):
                phase_wins += 1
    
    # 计算胜率，避免除零错误
    if phase_total == 0:
        win_rate = 0.5
    else:
        win_rate = phase_wins / phase_total
    
    # 使用相同的积分计算公式
    return self.calculate_points(win_rate, opponent_points, phase_total)
```

### 策略评估

评估指定策略的性能，生成全面的评估结果：

```python
def evaluate_strategy(self, strategy_name: str, strategy_fn: Callable, opponent_strategy: Callable, num_games: int = 50, opponent_points: int = None) -> Dict:
    """
    评估指定策略的性能
    
    Args:
        strategy_name: 策略名称
        strategy_fn: 策略函数
        opponent_strategy: 对手策略函数
        num_games: 对弈局数
        opponent_points: 对手积分
        
    Returns:
        评估结果
    """
    print(f"开始评估策略: {strategy_name}")
    
    results = {
        'strategy_name': strategy_name,
        'num_games': num_games,
        'wins': 0,          # 胜利局数
        'losses': 0,        # 失败局数
        'draws': 0,         # 平局局数
        'total_score': 0,   # 总得分
        'total_time': 0,    # 总时间
        'total_moves': 0,   # 总步数
        'game_details': [], # 游戏详情列表
        'opponent_points': opponent_points,
    }
    
    # 进行多局对弈，交替先手
    for game_idx in range(num_games):
        print(f"进行第 {game_idx + 1}/{num_games} 局对弈...")
        
        # 交替先手，确保公平性
        if game_idx % 2 == 0:
            winner, score, game_time, moves = self._play_game(strategy_fn, opponent_strategy, is_strategy_black=True)
        else:
            winner, score, game_time, moves = self._play_game(opponent_strategy, strategy_fn, is_strategy_black=False)
        
        # 记录游戏结果
        game_result = {
            'game_idx': game_idx,
            'winner': winner,
            'score': score,
            'game_time': game_time,
            'moves': moves,
            'is_strategy_black': game_idx % 2 == 0,
        }
        
        # 更新统计结果
        results['game_details'].append(game_result)
        results['total_score'] += score
        results['total_time'] += game_time
        results['total_moves'] += moves
        
        # 更新胜负统计
        if winner == 'black' and game_idx % 2 == 0:
            results['wins'] += 1
        elif winner == 'white' and game_idx % 2 == 1:
            results['wins'] += 1
        elif winner == 'draw':
            results['draws'] += 1
        else:
            results['losses'] += 1
    
    # 计算统计指标
    results['win_rate'] = results['wins'] / num_games
    results['average_score'] = results['total_score'] / num_games
    results['average_time'] = results['total_time'] / num_games
    results['average_moves'] = results['total_moves'] / num_games
    
    # 计算基于胜率的积分
    win_rate_points = self.win_rate_evaluator.calculate_points(
        win_rate=results['win_rate'],
        opponent_points=opponent_points,
        game_count=num_games
    )
    results['win_rate_points'] = win_rate_points
    
    # 计算各阶段胜率积分
    phase_results = self.win_rate_evaluator.evaluate_phases(results['game_details'], opponent_points)
    results['phase_results'] = phase_results
    
    # 计算稳定性评分
    stability_rating = self.win_rate_evaluator.calculate_stability_rating(num_games)
    results['stability_rating'] = stability_rating
    
    # 保存结果
    self.evaluation_results.append(results)
    self.game_history.extend(results['game_details'])
    
    return results
```

### 策略比较

比较多个策略的性能，生成对比结果：

```python
def compare_strategies(self, strategies: List[Tuple[str, Callable]], num_games: int = 30) -> Dict:
    """
    比较多个策略的性能
    
    Args:
        strategies: 策略列表，每个策略包含名称和函数
        num_games: 每对策略之间的对弈局数
        
    Returns:
        比较结果
    """
    comparison_results = {
        'strategies': [s[0] for s in strategies],
        'num_games_per_matchup': num_games,
        'matchup_results': [],
    }
    
    # 所有策略两两比较
    for i in range(len(strategies)):
        for j in range(i + 1, len(strategies)):
            strategy_a_name, strategy_a = strategies[i]
            strategy_b_name, strategy_b = strategies[j]
            
            print(f"\n比较策略: {strategy_a_name} vs {strategy_b_name}")
            
            # 策略A作为黑方
            results_a_black = self.evaluate_strategy(
                f"{strategy_a_name}_black",
                strategy_a,
                strategy_b,
                num_games=num_games // 2
            )
            
            # 策略B作为黑方
            results_b_black = self.evaluate_strategy(
                f"{strategy_b_name}_black",
                strategy_b,
                strategy_a,
                num_games=num_games // 2
            )
            
            # 合并结果
            total_wins_a = results_a_black['wins'] + results_b_black['losses']
            total_wins_b = results_a_black['losses'] + results_b_black['wins']
            total_draws = results_a_black['draws'] + results_b_black['draws']
            
            matchup_result = {
                'strategy_a': strategy_a_name,
                'strategy_b': strategy_b_name,
                'strategy_a_wins': total_wins_a,
                'strategy_b_wins': total_wins_b,
                'draws': total_draws,
                'strategy_a_win_rate': total_wins_a / num_games,
                'strategy_b_win_rate': total_wins_b / num_games,
                'draw_rate': total_draws / num_games,
            }
            
            comparison_results['matchup_results'].append(matchup_result)
    
    return comparison_results
```

```python
def analyze_strategy_performance(self, strategy_name: str, metrics: List[EvaluationMetric] = None) -> Dict:
    """
    分析指定策略的性能指标
    
    Args:
        strategy_name: 策略名称
        metrics: 要分析的指标列表
        
    Returns:
        分析结果
    """
    if metrics is None:
        metrics = list(EvaluationMetric)
    
    # 查找该策略的评估结果
    strategy_results = None
    for results in self.evaluation_results:
        if results['strategy_name'] == strategy_name:
            strategy_results = results
            break
    
    if not strategy_results:
        raise ValueError(f"未找到策略 {strategy_name} 的评估结果")
    
    analysis = {
        'strategy_name': strategy_name,
        'metrics': {},
    }
    
    # 计算各个指标
    if EvaluationMetric.WIN_RATE in metrics:
        analysis['metrics']['win_rate'] = strategy_results['win_rate']
    
    if EvaluationMetric.AVERAGE_SCORE in metrics:
        analysis['metrics']['average_score'] = strategy_results['average_score']
    
    if EvaluationMetric.AVERAGE_TIME in metrics:
        analysis['metrics']['average_time'] = strategy_results['average_time']
    
    if EvaluationMetric.MOVES_PER_GAME in metrics:
        analysis['metrics']['moves_per_game'] = strategy_results['average_moves']
    
    if EvaluationMetric.EXPLORATION_RATE in metrics:
        analysis['metrics']['exploration_rate'] = self._calculate_exploration_rate(strategy_results)
    
    if EvaluationMetric.EXPLOITATION_RATE in metrics:
        analysis['metrics']['exploitation_rate'] = self._calculate_exploitation_rate(strategy_results)
    
    if EvaluationMetric.TERRITORY_CONTROL in metrics:
        analysis['metrics']['territory_control'] = self._calculate_territory_control(strategy_results)
    
    if EvaluationMetric.MOVE_QUALITY in metrics:
        analysis['metrics']['move_quality'] = self._calculate_move_quality(strategy_results)
    
    if EvaluationMetric.STRATEGIC_CONSISTENCY in metrics:
        analysis['metrics']['strategic_consistency'] = self._calculate_strategic_consistency(strategy_results)
    
    if EvaluationMetric.ENDGAME_ACCURACY in metrics:
        analysis['metrics']['endgame_accuracy'] = self._calculate_endgame_accuracy(strategy_results)
    
    if EvaluationMetric.OPENING_QUALITY in metrics:
        analysis['metrics']['opening_quality'] = self._calculate_opening_quality(strategy_results)
    
    if EvaluationMetric.MIDDLE_GAME_QUALITY in metrics:
        analysis['metrics']['middle_game_quality'] = self._calculate_middle_game_quality(strategy_results)
    
    # 胜率积分相关指标
    if EvaluationMetric.WIN_RATE_POINTS in metrics:
        analysis['metrics']['win_rate_points'] = strategy_results.get('win_rate_points', self.win_rate_evaluator.base_points)
    
    if EvaluationMetric.OPENING_WIN_RATE_POINTS in metrics:
        analysis['metrics']['opening_win_rate_points'] = strategy_results.get('phase_results', {}).get('opening', {}).get('points', self.win_rate_evaluator.base_points)
    
    if EvaluationMetric.MIDDLE_WIN_RATE_POINTS in metrics:
        analysis['metrics']['middle_win_rate_points'] = strategy_results.get('phase_results', {}).get('middle', {}).get('points', self.win_rate_evaluator.base_points)
    
    if EvaluationMetric.ENDGAME_WIN_RATE_POINTS in metrics:
        analysis['metrics']['endgame_win_rate_points'] = strategy_results.get('phase_results', {}).get('endgame', {}).get('points', self.win_rate_evaluator.base_points)
    
    if EvaluationMetric.OPPONENT_ADJUSTED_POINTS in metrics:
        opponent_points = strategy_results.get('opponent_points', self.win_rate_evaluator.base_points)
        win_rate = strategy_results.get('win_rate', 0.5)
        adjusted_points = self.win_rate_evaluator.calculate_points(win_rate, opponent_points, strategy_results.get('num_games', 1))
        analysis['metrics']['opponent_adjusted_points'] = adjusted_points
    
    if EvaluationMetric.STABILITY_RATING in metrics:
        analysis['metrics']['stability_rating'] = strategy_results.get('stability_rating', 0.0)
    
    return analysis
```

### 探索率计算

衡量AI在搜索过程中探索新路径的程度：

```python
def _calculate_exploration_rate(self, results: Dict) -> float:
    """
    计算探索率
    
    Args:
        results: 评估结果
        
    Returns:
        探索率
    """
    # 探索率：基于游戏中选择的不同落子位置的多样性来计算
    if not results.get('game_details'):
        return 0.5
    
    total_unique_moves = 0
    total_moves = 0
    
    for game in results['game_details']:
        move_history = game.get('move_history', [])
        unique_moves = len(set(move_history))
        total_unique_moves += unique_moves
        total_moves += len(move_history)
    
    if total_moves == 0:
        return 0.0
    
    # 探索率 = 平均每局游戏的不同落子位置数 / 棋盘总位置数
    board_size = 19  # 假设标准19x19棋盘
    max_possible_moves = board_size * board_size
    exploration_rate = (total_unique_moves / len(results['game_details'])) / max_possible_moves
    
    return min(1.0, max(0.0, exploration_rate))
```

### 利用效率计算

衡量AI利用已知良好路径的程度：

```python
def _calculate_exploitation_rate(self, results: Dict) -> float:
    """
    计算利用率
    
    Args:
        results: 评估结果
        
    Returns:
        利用率
    """
    # 利用率：基于AI选择最佳落子的频率来计算
    if not results.get('game_details'):
        return 0.5
    
    total_exploit_moves = 0
    total_moves = 0
    
    for game in results['game_details']:
        best_move_history = game.get('best_move_history', [])
        actual_move_history = game.get('move_history', [])
        
        # 计算选择最佳落子的次数
        exploit_moves = sum(1 for actual, best in zip(actual_move_history, best_move_history) if actual == best)
        total_exploit_moves += exploit_moves
        total_moves += len(actual_move_history)
    
    if total_moves == 0:
        return 0.0
    
    # 利用率 = 选择最佳落子的频率
    exploitation_rate = total_exploit_moves / total_moves
    
    return min(1.0, max(0.0, exploitation_rate))
```

```python
def generate_performance_report(self, strategy_name: str, save_path: str = None) -> str:

    report += f"## 胜率趋势\n"
    report += "| 游戏局数 | 累计胜率 |\n"
    report += "|---------|---------|\n"
    
    cumulative_wins = 0
    for i, game in enumerate(strategy_results['game_details']):
        if game['winner'] == 'black' and game['is_strategy_black']:
            cumulative_wins += 1
        elif game['winner'] == 'white' and not game['is_strategy_black']:
            cumulative_wins += 1
        cumulative_win_rate = cumulative_wins / (i + 1)
        report += f"| {i + 1} | {cumulative_win_rate:.2%} |\n"
    
    # 生成积分趋势数据
    report += f"\n## 积分趋势\n"
    report += "| 游戏局数 | 胜率 | 累计积分 |\n"
    report += "|---------|------|----------|\n"
    
    cumulative_wins = 0
    for i, game in enumerate(strategy_results['game_details']):
        if game['winner'] == 'black' and game['is_strategy_black']:
            cumulative_wins += 1
        elif game['winner'] == 'white' and not game['is_strategy_black']:
            cumulative_wins += 1
        
        cumulative_win_rate = cumulative_wins / (i + 1)
        game_points = self.win_rate_evaluator.calculate_points(
            win_rate=cumulative_win_rate,
            opponent_points=strategy_results.get('opponent_points'),
            game_count=i + 1
        )
        report += f"| {i + 1} | {cumulative_win_rate:.2%} | {game_points:.0f} |\n"
    
    # 生成阶段积分对比
    report += f"\n## 阶段积分对比\n"
    report += "| 阶段 | 胜率 | 积分 | 局数 |\n"
    report += "|------|------|------|------|\n"
    
    phase_results = strategy_results.get('phase_results', {})
    for phase_name in ['opening', 'middle', 'endgame']:
        phase = phase_results.get(phase_name, {})
        win_rate = phase.get('win_rate', 0.5)
        points = phase.get('points', self.win_rate_evaluator.base_points)
        total_games = phase.get('total_games', 0)
        report += f"| {phase_name} | {win_rate:.2%} | {points:.0f} | {total_games} |\n"
    
    # 保存报告
    if save_path:
        with open(save_path, 'w', encoding='utf-8') as f:
            f.write(report)
        print(f"性能报告已保存到: {save_path}")
    
    return report
```

```python
def optimize_strategy(self, strategy_fn: Callable, evaluation_fn: Callable, initial_params: Dict, num_iterations: int = 100) -> Dict:
    """
    优化策略参数
    
    Args:
        strategy_fn: 策略函数
        evaluation_fn: 评估函数
        initial_params: 初始参数
        num_iterations: 迭代次数
        
    Returns:
        优化结果
    """
    print(f"开始优化策略参数，共 {num_iterations} 次迭代")
    
    best_params = initial_params.copy()
    best_score = evaluation_fn(strategy_fn, best_params)
    
    optimization_history = [
        {
            'iteration': 0,
            'params': best_params.copy(),
            'score': best_score,
        }
    ]
    
    for i in range(1, num_iterations + 1):
        # 生成新的参数
        new_params = self._generate_new_params(best_params)
        
        # 评估新参数
        new_score = evaluation_fn(strategy_fn, new_params)
        
        # 更新最佳参数
        if new_score > best_score:
            best_params = new_params.copy()
            best_score = new_score
            print(f"迭代 {i}/{num_iterations}: 找到更好的参数，得分: {best_score}")
        
        # 记录历史
        optimization_history.append({
            'iteration': i,
            'params': new_params.copy(),
            'score': new_score,
        })
    
    optimization_result = {
        'best_params': best_params,
        'best_score': best_score,
        'optimization_history': optimization_history,
    }
    
    return optimization_result
```
