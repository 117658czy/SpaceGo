### 2. 基准色驱动系统

核心是基于HSL色彩模型的智能配色算法：

```typescript
// 伪代码：基准色处理逻辑
class ColorGenerator {
  baseHue: number;        // 基准色相 (0-360)
  baseSaturation: number; // 基准饱和度 (0-100%)
  baseLightness: number;  // 基准明度 (0-100%)
  
  generateColorPalette(baseColor: string): ColorScheme {
    // 1. 将基准色转换为HSL值
    const hsl = this.rgbToHsl(baseColor);
    
    // 2. 生成相邻色相 (±15°, ±30°)
    const analogousColors = this.generateAnalogous(hsl, 15, 30);
    
    // 3. 生成深浅变体 (明度 ±20%, ±40%)
    const lightnessVariants = this.generateLightnessVariants(hsl, 20, 40);
    
    // 4. 生成饱和度变体 (饱和度 ±20%, ±40%)
    const saturationVariants = this.generateSaturationVariants(hsl, 20, 40);
    
    return this.combineColorGroups(analogousColors, lightnessVariants, saturationVariants);
  }
}
```

class SmartColorAdjuster {
  // 确保颜色在可访问性范围内
  ensureAccessibility(foreground: string, background: string): boolean {
    const contrast = this.calculateContrast(foreground, background);
    return contrast >= 4.5; // WCAG AA标准
  }
  
  // 生成和谐的颜色组合
  generateHarmoniousColors(baseHue: number): string[] {
    // 使用色相环的黄金分割点 (137.5°)
    return [
      baseHue,
      (baseHue + 137.5) % 360,
      (baseHue + 275) % 360
    ];
  }
}
